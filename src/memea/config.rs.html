1//! Configuration management for MemEA memory peripheral estimation.
2//!
3//! This module provides functionality to read and manage memory configurations
4//! from YAML files. Each configuration specifies memory array parameters,
5//! cell types, voltages, and ADC settings used for peripheral estimation.
6
7use serde::Deserialize;
8use std::fs;
9use std::io::BufReader;
10use std::{collections::HashMap, path::PathBuf};
11use thiserror::Error;
12
13use crate::{errorln, Float, MemeaError};
14
15/// A collection of memory configurations indexed by name.
16type Configs = HashMap<String, Config>;
17
18/// Errors that can occur during configuration processing.
19#[derive(Debug, Error)]
20pub enum ConfigError {
21    /// Indicates that a required configuration option was not found.
22    #[error("Cannot find option in config: {0}")]
23    MissingOption(String),
24}

26/// Represents one memory configuration for peripheral estimation.
27///
28/// This struct is typically deserialized from YAML or JSON using `serde`. It
29/// contains both essential and optional parameters that control the peripheral
30/// estimation process.
31///
32/// # Examples
33///
34/// ```yaml
35/// name: 64-64
36/// n: 128
37/// m: 64
38/// bl: [1, 2, 0]
39/// wl: [4, 2.5, 0, 1]
40/// well: [0, 4]
41/// cell: 1FeFET_100
42/// enob: 1
43/// fs: 1e9
44/// adcs: 64
45/// ```
46#[derive(Debug, Deserialize)]
47pub struct Config {
48    /// Name of the configuration. If not supplied, the file path will be used.
49    pub name: Option<String>,
50
51    /// Number of rows in the memory array.
52    pub n: usize,
53    /// Number of columns in the memory array.
54    pub m: usize,
55    /// Memory cell type to use for estimation.
56    pub cell: String,
57
58    /// Bitline voltages
59    pub bl: Option<Vec<Float>>,
60    /// Wordline voltages
61    pub wl: Option<Vec<Float>>,
62    /// Voltages required for well biasing
63    pub well: Option<Vec<Float>>,
64
65    /// Array processing rate / ADC sampling rate
66    pub fs: Float,
67
68    /// Number of downstream analog-to-digital converters.
69    pub adcs: Option<usize>,
70    /// Number of bits required for ADCs.
71    pub bits: Option<usize>,
72
73    /// Additional configuration options as key-value pairs.
74    pub options: Option<HashMap<String, String>>,
75}
76
77/// Deserializes a configuration from a YAML file.
78///
79/// # Arguments
80/// * `filename` - Path of the YAML file to read
81///
82/// # Returns
83/// * `Ok(Config)` - Successfully parsed configuration
84/// * `Err(MemeaError)` - File I/O error or YAML parsing error
85///
86/// # Examples
87/// ```no_run
88/// use std::path::PathBuf;
89/// # use memea::config::read;
90///
91/// let config_path = PathBuf::from("config.yaml");
92/// let config = read(&config_path).expect("Failed to read config");
93/// ```
94fn read(filename: &std::path::PathBuf) -> Result<Config, MemeaError> {
95    // TODO: Check filetype
96    let file = fs::File::open(filename)?;
97    let rdr = BufReader::new(file);
98    let config: Config = serde_yaml::from_reader(rdr)?;
99
100    Ok(config)
101}

103/// Reads multiple configuration files and returns them indexed by name.
104///
105/// This function attempts to read all provided configuration files. If a file
106/// fails to parse, an error is logged and that file is skipped. The resulting
107/// HashMap uses either the configured name or the file path as the key.
108///
109/// # Arguments
110/// * `paths` - Vector of configuration file paths to read
111///
112/// # Returns
113/// * `HashMap<String, Config>` - Successfully parsed configurations indexed by name
114///
115/// # Examples
116/// ```no_run
117/// use std::path::PathBuf;
118/// # use memea::config::read_all;
119///
120/// let paths = vec![
121///     PathBuf::from("config1.yaml"),
122///     PathBuf::from("config2.yaml"),
123/// ];
124/// let configs = read_all(&paths);
125/// println!("Loaded {} configurations", configs.len());
126/// ```
127pub fn read_all(paths: &Vec<PathBuf>) -> Configs {
128    let mut configs: Configs = HashMap::new();
129    for c in paths {
130        match read(c) {
131            Ok(r) => {
132                let name = match &r.name {
133                    Some(s) => s.clone(),
134                    None => c.to_string_lossy().into(),
135                };
136
137                configs.insert(name, r);
138            }
139            Err(e) => errorln!("Failed to read config {:?} ({})", &c, e),
140        }
141    }
142
143    configs
144}
